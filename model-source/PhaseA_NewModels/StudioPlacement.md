local function part(model, name, pos, size)
    local p = Instance.new("Part")
    p.Name, p.Size, p.CFrame = name, Vector3.new(table.unpack(size)), CFrame.new(Vector3.new(table.unpack(pos)))
    p.Transparency, p.Anchored = 1, true
    p.CanCollide, p.CanTouch, p.CanQuery = false, false, false
    p.Parent = model
    return p
end
for _, imported in Selection:Get() do
    local spec = specs[imported.Name]
    if not imported:IsA("Model") or not spec then
        warn("Skip unrecognized selected model", imported.Name)
        continue
    end
    local destination = assets:FindFirstChild(spec.folder)
    if destination and destination:FindFirstChild(spec.name) then
        warn("Preserved existing destination model", spec.folder, spec.name)
        continue
    end
    local parts = {}
    local rejected = false
    for _, d in imported:GetDescendants() do
        if d:IsA("LuaSourceContainer") then rejected = true end
        if d:IsA("BasePart") then table.insert(parts, d) end
    end
    if rejected or #parts == 0 then warn("Skip unexpected import contents", imported.Name) continue end
    -- Check the visible source bounds before adding invisible primary parts.
    local _, size = imported:GetBoundingBox()
    local expected = Vector3.new(table.unpack(spec.dimensions_studs))
    local relativeError = (size - expected).Magnitude / math.max(expected.Magnitude, 0.01)
    if relativeError > 0.12 then
        warn("Scale/orientation mismatch; fix import first", imported.Name, size, expected)
        continue
    end
    -- Bounds alone cannot prove pose, origin or ownership; verify those before running.
    local model = Instance.new("Model")
    model.Name = spec.name
    local primary = part(model, spec.primary_part, spec.primary_position_studs,
        spec.primary_part == "Handle" and {0.4, 0.4, 0.4} or {1, 1, 1})
    model.PrimaryPart = primary
    for _, p in parts do
        p.Anchored = true
        local name = string.lower(p.Name)
        p.CanCollide = spec.name == "Workbench" and (string.find(name, "worktop") ~= nil or string.find(name, "leg") ~= nil)
        p.CanTouch, p.CanQuery = false, false
        p.Parent = model
    end
    for _, marker in spec.markers do
        local p = part(model, marker.name, marker.position_studs, marker.size_studs)
        if marker.Slots then p:SetAttribute("Slots", marker.Slots) end
    end
    local panel = nil
    for _, p in parts do if string.find(p.Name, "__Body", 1, true) then panel = p break end end

    for i, pos in spec.slots do
        local a = Instance.new("Attachment")
        a.Name, a.Parent = "Slot" .. i, panel
        a.WorldCFrame = CFrame.new(Vector3.new(table.unpack(pos)))
    end
    if not destination then
        destination = Instance.new("Folder")
        destination.Name, destination.Parent = spec.folder, assets
    end
    model.Parent = destination
    imported:Destroy()
    print("Placed", spec.folder, spec.name, "primary", primary.Name, "slots", #spec.slots)
